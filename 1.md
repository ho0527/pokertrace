# 德州撲克手牌紀錄系統 - 技術架構與實作建議

## 🏗️ **技術棧架構**

### **前端技術棧**
- **HTML5 + CSS3 + JavaScript (ES6+)**
- **TailwindCSS** - 現代化樣式框架
- **ECharts 5.x** - 數據視覺化圖表庫
- **Alpine.js** (可選) - 輕量級前端響應式框架

### **後端技術棧**
- **Django 4.x + Django REST Framework**
- **PostgreSQL 14+** - 主資料庫
- **Redis** (建議) - 緩存和會話存儲

### **ECharts 整合方案**

```javascript
// static/js/charts.js
class PokerCharts {
    constructor() {
        this.theme = 'light';  // 或 'dark'
    }

    // 統計圖表配置
    createStatChart(elementId, data, type = 'line') {
        const chart = echarts.init(document.getElementById(elementId));
        
        const option = {
            title: {
                text: data.title,
                textStyle: {
                    color: '#374151'
                }
            },
            tooltip: {
                trigger: 'axis',
                formatter: this.formatTooltip
            },
            xAxis: {
                type: 'category',
                data: data.labels
            },
            yAxis: {
                type: 'value'
            },
            series: [{
                data: data.values,
                type: type,
                smooth: true,
                itemStyle: {
                    color: '#22c55e'
                }
            }]
        };
        
        chart.setOption(option);
        return chart;
    }

    // VPIP/PFR 雷達圖
    createPlayerRadar(elementId, playerData) {
        const chart = echarts.init(document.getElementById(elementId));
        
        const option = {
            title: {
                text: '玩家統計雷達圖'
            },
            radar: {
                indicator: [
                    { name: 'VPIP', max: 100 },
                    { name: 'PFR', max: 100 },
                    { name: '3-Bet', max: 20 },
                    { name: 'C-Bet', max: 100 },
                    { name: 'WTSD', max: 50 },
                    { name: 'W$SD', max: 100 }
                ]
            },
            series: [{
                type: 'radar',
                data: [{
                    value: playerData,
                    name: '統計數據'
                }]
            }]
        };
        
        chart.setOption(option);
        return chart;
    }

    // 盈虧趋势图
    createProfitChart(elementId, profitData) {
        const chart = echarts.init(document.getElementById(elementId));
        
        const option = {
            title: {
                text: '盈虧趨勢'
            },
            xAxis: {
                type: 'category',
                data: profitData.dates
            },
            yAxis: {
                type: 'value',
                axisLabel: {
                    formatter: 'NT${value}'
                }
            },
            series: [{
                data: profitData.amounts,
                type: 'line',
                markLine: {
                    data: [
                        { type: 'average', name: '平均值' }
                    ]
                },
                areaStyle: {
                    opacity: 0.3
                }
            }]
        };
        
        chart.setOption(option);
        return chart;
    }

    formatTooltip(params) {
        // 自定義提示框格式
        return `${params[0].name}: ${params[0].value}%`;
    }
}

// 全域初始化
const pokerCharts = new PokerCharts();
```

### **ViewSet 範例**

```python
# apps/hands/views.py
from rest_framework import viewsets, status
from rest_framework.decorators import action
from rest_framework.response import Response
from django_filters.rest_framework import DjangoFilterBackend

class HandViewSet(viewsets.ModelViewSet):
    queryset = Hand.objects.all()
    serializer_class = HandSerializer
    filter_backends = [DjangoFilterBackend]
    filterset_fields = ['session', 'state', 'created_at']
    
    @action(detail=True, methods=['get'])
    def replay_data(self, request, pk=None):
        """獲取手牌重播數據"""
        hand = self.get_object()
        replay_data = hand.get_replay_data()
        return Response(replay_data)
    
    @action(detail=False, methods=['get'])
    def statistics(self, request):
        """手牌統計數據"""
        stats = self.get_queryset().aggregate(
            total_hands=Count('id'),
            avg_pot=Avg('total_pot'),
            total_rake=Sum('rake')
        )
        return Response(stats)
```

## 📊 **前端數據流程**

### **AJAX 數據請求範例**

```javascript
// static/js/api.js
class PokerAPI {
    constructor() {
        this.baseURL = '/api/v1/';
        this.csrfToken = this.getCSRFToken();
    }

    async getHandHistory(filters = {}) {
        const params = new URLSearchParams(filters);
        const response = await fetch(`${this.baseURL}hands/?${params}`, {
            headers: {
                'X-CSRFToken': this.csrfToken,
                'Content-Type': 'application/json'
            }
        });
        return await response.json();
    }

    async getPlayerStats(playerId, dateRange = {}) {
        const params = new URLSearchParams(dateRange);
        const response = await fetch(`${this.baseURL}players/${playerId}/statistics/?${params}`);
        return await response.json();
    }

    async recordHandAction(handId, actionData) {
        const response = await fetch(`${this.baseURL}hands/${handId}/actions/`, {
            method: 'POST',
            headers: {
                'X-CSRFToken': this.csrfToken,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify(actionData)
        });
        return await response.json();
    }

    getCSRFToken() {
        return document.querySelector('[name=csrfmiddlewaretoken]').value;
    }
}

const api = new PokerAPI();
```
